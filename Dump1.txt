-- CE Lua: FAST nodePoolBase finder by clustered stride
-- READ ONLY.
-- Не патчит память.
--
-- Исправляет проблему зависания:
-- старый скрипт делал 13k+ raw candidates и пытался score'ить все.
-- этот скрипт сначала группирует по stride 0x158 и проверяет только сильные базы.

------------------------------------------------------------
-- CONFIG
------------------------------------------------------------

local CONFIG = {
    stride = 0x158,

    -- Самые вероятные offsets float pattern 1.0,2.0,1.0,1.0 внутри node.
    -- Старый удачный дамп указывал примерно на +0x140.
    possibleFloatOffsets = {
        0x130, 0x138, 0x140, 0x148, 0x150
    },

    -- Сколько node назад пробовать от каждого float-hit.
    maxStrideBack = 20,

    -- Минимум повторений в одной предполагаемой базе.
    -- Если слишком мало кандидатов — поставь 3.
    -- Если слишком много — поставь 5 или 6.
    minClusterHits = 4,

    -- Сколько лучших кандидатов вообще score'ить.
    maxScoredCandidates = 120,

    -- Сколько node'ов проверять по stride.
    checkNodes = 10,

    -- Верхний лимит времени на scoring.
    timeBudgetSeconds = 20,

    -- Минимальный score для вывода.
    minScore = 420,

    -- Сколько результатов вывести.
    maxPrint = 30,

    addTopToAddressList = 8,

    protectionFlags = "+W-C",
}

------------------------------------------------------------
-- LOG
------------------------------------------------------------

local function makeOutPath()
    local base = os.getenv("USERPROFILE") or os.getenv("HOME") or "."
    return base .. "\\Desktop\\ce_nodePoolBase_FAST_cluster_" .. os.date("%Y%m%d_%H%M%S") .. ".txt"
end

local OUT = makeOutPath()
local F = io.open(OUT, "w")
if not F then error("Cannot create output file: " .. OUT) end

local function log(s)
    s = tostring(s or "")
    print(s)
    F:write(s .. "\n")
    F:flush()
end

------------------------------------------------------------
-- UTILS
------------------------------------------------------------

local function hex(n)
    if not n then return "nil" end
    return string.format("0x%X", n)
end

local function alignDown(v, a)
    return v - (v % a)
end

local function safeReadBytes(addr, count)
    local ok, b = pcall(readBytes, addr, count, true)
    if ok then return b end
    return nil
end

local function safeReadQword(addr)
    local ok, v = pcall(readQword, addr)
    if ok then return v end
    return nil
end

local function safeReadInteger(addr)
    local ok, v = pcall(readInteger, addr)
    if ok then return v end
    return nil
end

local function safeReadFloat(addr)
    local ok, v = pcall(readFloat, addr)
    if ok then return v end
    return nil
end

local function safeReadString(addr, len, wide)
    local ok, s = pcall(readString, addr, len, wide)
    if ok and s then return s end
    return nil
end

local function isReadable(addr)
    return addr and addr > 0x10000 and safeReadBytes(addr, 1) ~= nil
end

-- Быстрая проверка pointer-like без чтения по указателю.
-- Старый isProbablyPtr делал readBytes(ptr,1), это сильно тормозило.
local function isLikelyPtr(v)
    if not v then return false end
    if v < 0x10000 then return false end
    if v > 0x0000800000000000 then return false end
    return true
end

local function low32(v)
    if not v then return nil end
    return v % 0x100000000
end

local function near(v, t, eps)
    if not v then return false end
    return math.abs(v - t) <= eps
end

local function addrFromString(s)
    local ok, a = pcall(getAddressSafe, s)
    if ok and a then return a end
    return tonumber(tostring(s):gsub("^0x", ""), 16)
end

local function addRecord(desc, addr)
    local al = getAddressList()
    local mr = al.createMemoryRecord()
    mr.Description = desc
    mr.Address = hex(addr)
    mr.Type = vtQword
    mr.ShowAsHex = true
end

local function hasFloat1211At(addr)
    local f1 = safeReadFloat(addr)
    local f2 = safeReadFloat(addr + 4)
    local f3 = safeReadFloat(addr + 8)
    local f4 = safeReadFloat(addr + 12)

    return near(f1, 1.0, 0.001)
       and near(f2, 2.0, 0.001)
       and near(f3, 1.0, 0.001)
       and near(f4, 1.0, 0.001)
end

local function readPtrText(ptr)
    if not isLikelyPtr(ptr) then return nil end

    local ascii = safeReadString(ptr, 96, false)
    local wide  = safeReadString(ptr, 96, true)

    if ascii and #ascii > 0 and string.match(ascii, "^[%w_ /%-%.]+$") then
        return ascii
    end

    if wide and #wide > 0 then return wide end
    if ascii and #ascii > 0 then return ascii end

    return nil
end

local function readFieldText(addr)
    local p = safeReadQword(addr)
    if not isLikelyPtr(p) then return p, nil end
    return p, readPtrText(p)
end

local function aobScanFast(aob)
    local ok, sl = pcall(function()
        return AOBScan(aob, CONFIG.protectionFlags)
    end)

    if ok and sl then
        return sl, "AOBScan with flags " .. CONFIG.protectionFlags
    end

    local sl2 = AOBScan(aob)
    return sl2, "AOBScan without flags"
end

------------------------------------------------------------
-- SCORING
------------------------------------------------------------

local function scoreHeader(node, first)
    if not isReadable(node) then
        return 0
    end

    local score = 0

    local q00 = safeReadQword(node + 0x00)
    local q08 = safeReadQword(node + 0x08)
    local q10 = safeReadQword(node + 0x10)
    local i18 = safeReadInteger(node + 0x18)
    local q28 = safeReadQword(node + 0x28)
    local q30 = safeReadQword(node + 0x30)

    if isLikelyPtr(q00) then score = score + 18 end
    if isLikelyPtr(q08) then score = score + 8 end
    if isLikelyPtr(q10) then score = score + 18 end
    if i18 and i18 >= 0 and i18 <= 5 then score = score + 16 end
    if isLikelyPtr(q28) then score = score + 8 end
    if isLikelyPtr(q30) then score = score + 8 end

    if first then
        if first.q00 and q00 == first.q00 then score = score + 15 end
        if first.q08 and q08 == first.q08 then score = score + 8 end
        if first.q10 and q10 == first.q10 then score = score + 15 end
        if first.q28 and q28 == first.q28 then score = score + 6 end
        if first.q30 and q30 == first.q30 then score = score + 6 end
    end

    return score
end

local function scoreCandidate(base, clusterHits)
    if not isReadable(base) then
        return 0, "unreadable", {}
    end

    local score = 0
    local notes = {}
    local details = {}

    score = score + clusterHits * 90
    table.insert(notes, "clusterHits=" .. tostring(clusterHits))

    local first = {
        q00 = safeReadQword(base + 0x00),
        q08 = safeReadQword(base + 0x08),
        q10 = safeReadQword(base + 0x10),
        q28 = safeReadQword(base + 0x28),
        q30 = safeReadQword(base + 0x30),
    }

    local goodHeaders = 0

    for i = 0, CONFIG.checkNodes - 1 do
        local node = base + i * CONFIG.stride
        local hs = scoreHeader(node, first)

        if hs >= 55 then
            goodHeaders = goodHeaders + 1
        end

        score = score + hs

        if i < 6 then
            table.insert(details, {
                kind = "header",
                index = i,
                node = node,
                headerScore = hs
            })
        end
    end

    table.insert(notes, "goodHeaders=" .. tostring(goodHeaders))

    if goodHeaders >= 4 then
        score = score + 120
        table.insert(notes, "strideOK")
    end

    if goodHeaders >= 8 then
        score = score + 220
        table.insert(notes, "strongStride")
    end

    -- Лёгкая проверка count20 / float1211 на первых node'ах.
    local count20Nodes = 0
    local floatNodes = 0

    for i = 0, math.min(CONFIG.checkNodes - 1, 8) do
        local node = base + i * CONFIG.stride

        for off = 0x100, 0x170, 8 do
            local q = safeReadQword(node + off)
            if low32(q) == 20 then
                count20Nodes = count20Nodes + 1
                break
            end
        end

        for off = 0x120, 0x160, 4 do
            if hasFloat1211At(node + off) then
                floatNodes = floatNodes + 1
                break
            end
        end
    end

    if count20Nodes > 0 then
        score = score + count20Nodes * 35
        table.insert(notes, "count20Nodes=" .. tostring(count20Nodes))
    end

    if floatNodes > 0 then
        score = score + floatNodes * 45
        table.insert(notes, "float1211Nodes=" .. tostring(floatNodes))
    end

    -- Чтение key/value только после дешёвых проверок.
    -- Это дорогая часть, поэтому делаем её только если база уже похожа.
    if score >= 350 then
        local known = {
            { name = "LevelTextNode",         off = 0x968 },
            { name = "DefaultWasTakenNode",   off = 0xD70 },
            { name = "DefaultTakeButtonNode", off = 0xEC8 },
        }

        for _, k in ipairs(known) do
            local node = base + k.off
            local keyPtr, keyText = readFieldText(node + 0x80)
            local valPtr, valText = readFieldText(node + 0xF8)

            table.insert(details, {
                kind = "known",
                name = k.name,
                node = node,
                keyText = keyText,
                valText = valText,
            })

            if keyText then
                if string.find(keyText, "LevelText", 1, true) then
                    score = score + 250
                    table.insert(notes, "LevelText.key")
                end
                if string.find(keyText, "DefaultWasTakenText", 1, true) then
                    score = score + 250
                    table.insert(notes, "DefaultWasTaken.key")
                end
                if string.find(keyText, "DefaultTakeButtonText", 1, true) then
                    score = score + 250
                    table.insert(notes, "DefaultTakeButton.key")
                end
            end

            if valText then
                if string.find(valText, "Получено", 1, true) then
                    score = score + 140
                    table.insert(notes, "val=Получено")
                end
                if string.find(valText, "Take", 1, true) then
                    score = score + 100
                    table.insert(notes, "val=Take")
                end
                if string.find(valText, "Забрать", 1, true) then
                    score = score + 140
                    table.insert(notes, "val=Забрать")
                end
            end
        end
    end

    return score, table.concat(notes, " | "), details
end

------------------------------------------------------------
-- MAIN
------------------------------------------------------------

log("FAST clustered nodePoolBase finder / READ ONLY")
log("Date: " .. os.date("%Y-%m-%d %H:%M:%S"))
log("Output: " .. OUT)
log("")
log("Перед запуском открой BattlePass/Rewards UI.")
log("")

local FLOAT_AOB = "00 00 80 3F 00 00 00 40 00 00 80 3F 00 00 80 3F"

log("============================================================")
log("STEP 1: FLOAT 1,2,1,1 SCAN")
log("============================================================")
log("AOB=" .. FLOAT_AOB)

local sl, mode = aobScanFast(FLOAT_AOB)
log("scan mode: " .. mode)

if not sl then
    log("hits=0")
    log("DONE")
    F:close()
    messageDialog("No float hits.\n\nReport:\n" .. OUT, mtInformation, mbOK)
    return
end

local hitCount = sl.Count
log("hits=" .. tostring(hitCount))

local hits = {}
for i = 0, hitCount - 1 do
    local h = addrFromString(sl[i])
    if h then table.insert(hits, h) end
end
sl.destroy()

------------------------------------------------------------
-- STEP 2: CLUSTER ONLY, NO HEAVY SCORE
------------------------------------------------------------

log("")
log("============================================================")
log("STEP 2: BUILD STRIDE CLUSTERS")
log("============================================================")

local clusterMap = {}

local function addBase(base, reason)
    if not base or base < 0x10000 then return end
    base = alignDown(base, 0x8)

    local key = hex(base)
    if not clusterMap[key] then
        clusterMap[key] = {
            base = base,
            hitCount = 0,
            reasons = {},
        }
    end

    clusterMap[key].hitCount = clusterMap[key].hitCount + 1

    if #clusterMap[key].reasons < 5 then
        table.insert(clusterMap[key].reasons, reason)
    end
end

for _, hit in ipairs(hits) do
    for _, off in ipairs(CONFIG.possibleFloatOffsets) do
        local nodeGuess = hit - off

        for back = 0, CONFIG.maxStrideBack do
            local base = nodeGuess - back * CONFIG.stride
            addBase(base, "hit=" .. hex(hit) .. " off=0x" .. string.format("%X", off) .. " back=" .. tostring(back))
        end
    end
end

local clusters = {}
for _, c in pairs(clusterMap) do
    if c.hitCount >= CONFIG.minClusterHits and isReadable(c.base) then
        table.insert(clusters, c)
    end
end

table.sort(clusters, function(a, b)
    if a.hitCount == b.hitCount then return a.base < b.base end
    return a.hitCount > b.hitCount
end)

log("clusters after prefilter=" .. tostring(#clusters))
log("will score max=" .. tostring(CONFIG.maxScoredCandidates))

------------------------------------------------------------
-- STEP 3: SCORE ONLY TOP CLUSTERS
------------------------------------------------------------

log("")
log("============================================================")
log("STEP 3: SCORE TOP CLUSTERS")
log("============================================================")

local accepted = {}
local startClock = os.clock()

local toScore = math.min(#clusters, CONFIG.maxScoredCandidates)

for i = 1, toScore do
    if os.clock() - startClock > CONFIG.timeBudgetSeconds then
        log("time budget reached; stop scoring")
        break
    end

    local c = clusters[i]
    local score, notes, details = scoreCandidate(c.base, c.hitCount)

    if score >= CONFIG.minScore then
        c.score = score
        c.notes = notes
        c.details = details
        table.insert(accepted, c)
    end

    if i % 20 == 0 then
        log("scored " .. tostring(i) .. "/" .. tostring(toScore))
    end
end

table.sort(accepted, function(a, b)
    if a.score == b.score then return a.base < b.base end
    return a.score > b.score
end)

------------------------------------------------------------
-- RESULTS
------------------------------------------------------------

log("")
log("============================================================")
log("RESULTS")
log("============================================================")
log("accepted=" .. tostring(#accepted))

for i, c in ipairs(accepted) do
    if i > CONFIG.maxPrint then break end

    log("")
    log(string.format(
        "#%02d nodePoolBase=%s score=%d clusterHits=%d",
        i,
        hex(c.base),
        c.score,
        c.hitCount
    ))

    log("notes: " .. tostring(c.notes))

    log("derived:")
    log("  LevelTextNode          = " .. hex(c.base + 0x968))
    log("  DefaultWasTakenNode    = " .. hex(c.base + 0xD70))
    log("  DefaultTakeButtonNode  = " .. hex(c.base + 0xEC8))

    log("reasons:")
    for _, r in ipairs(c.reasons or {}) do
        log("  " .. r)
    end

    log("details:")
    for _, d in ipairs(c.details or {}) do
        if d.kind == "header" then
            log(string.format(
                "  node[%s] %s headerScore=%s",
                tostring(d.index),
                hex(d.node),
                tostring(d.headerScore)
            ))
        elseif d.kind == "known" then
            log(string.format(
                "  %-24s node=%s key=%s val=%s",
                tostring(d.name),
                hex(d.node),
                tostring(d.keyText),
                tostring(d.valText)
            ))
        end
    end

    if i <= CONFIG.addTopToAddressList then
        addRecord("[FAST nodePoolBase] #" .. tostring(i), c.base)
        addRecord("[FAST nodePoolBase] #" .. tostring(i) .. " LevelTextNode", c.base + 0x968)
        addRecord("[FAST nodePoolBase] #" .. tostring(i) .. " DefaultWasTakenNode", c.base + 0xD70)
        addRecord("[FAST nodePoolBase] #" .. tostring(i) .. " DefaultTakeButtonNode", c.base + 0xEC8)
    end
end

log("")
log("DONE")
log("Report: " .. OUT)

F:close()

messageDialog(
    "FAST clustered scan complete.\n\nAccepted: " .. tostring(#accepted) ..
    "\n\nReport:\n" .. OUT,
    mtInformation,
    mbOK
)
