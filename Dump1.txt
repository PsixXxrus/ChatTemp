-- CE Lua: find useful base -> main dump -> outbound pointer micro-dumps
-- READ ONLY.
--
-- Тактика:
-- 1) найти первый полезный text-node base через старый float/stride подход;
-- 2) снять дамп base-512..base+512;
-- 3) найти QWORD pointers, которые ведут в другие области памяти;
-- 4) сделать micro dump ptr-64..ptr+64;
-- 5) вывести рейтинг интересных указателей.

------------------------------------------------------------
-- CONFIG
------------------------------------------------------------

local CONFIG = {
    -- Если хочешь отключить авто-поиск и руками задать base:
    -- MANUAL_BASE = 0x123456789AB,
    MANUAL_BASE = nil,

    floatAob = "00 00 80 3F 00 00 00 40 00 00 80 3F 00 00 80 3F",
    protectionFlags = "+W-C",

    stride = 0x158,

    possibleFloatOffsets = {
        0x130, 0x138, 0x140, 0x148, 0x150
    },

    maxStrideBack = 24,
    minVoteHits = 4,
    maxCandidatesToScore = 160,
    checkNodes = 10,

    mainBack = 0x200, -- -512
    mainFwd  = 0x200, -- +512

    microBack = 0x40, -- -64
    microFwd  = 0x40, -- +64

    maxMicroDumps = 80,

    keywords = {
        "DefaultTakeButton",
        "DefaultTakeButtonText",
        "DefaultWasTaken",
        "DefaultWasTakenText",
        "DefaultBlockPart",
        "BlockImg",
        "RewardGrid",
        "LevelText",
        "BattlePass",
        "Reward",
        "Take",
        "Забрать",
        "Получено",
        "Combat Knife",
        "Tactical Kukri",
        "Boxed 12ga",
    },
}

-- LOG

local function makeOutPath()
    local base = os.getenv("USERPROFILE") or os.getenv("HOME") or "."
    return base .. "\\Desktop\\ce_base_outbound_microdumps_" .. os.date("%Y%m%d_%H%M%S") .. ".txt"
end

local OUT = makeOutPath()
local F = io.open(OUT, "w")
if not F then error("Cannot create output file: " .. OUT) end

local function log(s)
    s = tostring(s or "")
    print(s)
    F:write(s .. "\n")
    F:flush()
end

------------------------------------------------------------
-- UTILS
------------------------------------------------------------

local function hex(n)
    if n == nil then return "nil" end
    return string.format("0x%X", n)
end

local function alignDown(v, a)
    return v - (v % a)
end

local function safeReadBytes(addr, count)
    local ok, b = pcall(readBytes, addr, count, true)
    if ok then return b end
    return nil
end

local function safeReadQword(addr)
    local ok, v = pcall(readQword, addr)
    if ok then return v end
    return nil
end

local function safeReadInteger(addr)
    local ok, v = pcall(readInteger, addr)
    if ok then return v end
    return nil
end

local function safeReadFloat(addr)
    local ok, v = pcall(readFloat, addr)
    if ok then return v end
    return nil
end

local function safeReadString(addr, len, wide)
    local ok, s = pcall(readString, addr, len, wide)
    if ok and s then return s end
    return nil
end

local function isReadable(addr)
    return addr and addr > 0x10000 and safeReadBytes(addr, 1) ~= nil
end

local function isLikelyPtr(v)
    if not v then return false end
    if v < 0x10000 then return false end
    if v > 0x0000800000000000 then return false end
    return true
end

local function isModulePtr(v)
    return v and v >= 0x00007FF000000000 and v < 0x0000800000000000
end

local function low32(v)
    if not v then return nil end
    return v % 0x100000000
end

local function near(v, target, eps)
    if not v then return false end
    return math.abs(v - target) <= eps
end

local function addrFromString(s)
    local ok, a = pcall(getAddressSafe, s)
    if ok and a then return a end
    return tonumber(tostring(s):gsub("^0x", ""), 16)
end

local function aobScanFast(aob)
    local ok, sl = pcall(function()
        return AOBScan(aob, CONFIG.protectionFlags)
    end)

    if ok and sl then
        return sl, "AOBScan with flags " .. CONFIG.protectionFlags
    end

    return AOBScan(aob), "AOBScan without flags"
end

local function addRecord(desc, addr)
    local al = getAddressList()
    local mr = al.createMemoryRecord()
    mr.Description = desc
    mr.Address = hex(addr)
    mr.Type = vtQword
    mr.ShowAsHex = true
end

local function isPrintable(b)
    return b >= 0x20 and b <= 0x7E
end

local function bytesToHexAscii(bytes, addr)
    local out = {}
    local i = 1

    while i <= #bytes do
        local hs, as = {}, {}

        for j = 0, 15 do
            local b = bytes[i + j]
            if b == nil then
                table.insert(hs, "  ")
                table.insert(as, " ")
            else
                table.insert(hs, string.format("%02X", b))
                table.insert(as, isPrintable(b) and string.char(b) or ".")
            end
        end

        table.insert(out, string.format(
            "%s : %-47s | %s",
            hex(addr + i - 1),
            table.concat(hs, " "),
            table.concat(as, "")
        ))

        i = i + 16
    end

    return table.concat(out, "\n")
end

local function dumpRange(addr, size, title)
    log("")
    log("------------------------------------------------------------")
    log(title .. " " .. hex(addr) .. " .. " .. hex(addr + size))
    log("------------------------------------------------------------")

    local bytes = safeReadBytes(addr, size)

    if bytes then
        log(bytesToHexAscii(bytes, addr))
    else
        log("[unreadable]")
    end
end

local function readPtrText(ptr)
    if not isLikelyPtr(ptr) then return nil end

    local ascii = safeReadString(ptr, 128, false)
    local wide  = safeReadString(ptr, 128, true)

    if ascii and #ascii > 0 and string.match(ascii, "^[%w_ /%-%.:]+$") then
        return ascii
    end

    if wide and #wide > 0 then return wide end
    if ascii and #ascii > 0 then return ascii end

    return nil
end

local function readFieldText(addr)
    local p = safeReadQword(addr)
    if not isLikelyPtr(p) then return p, nil end
    return p, readPtrText(p)
end

local function containsKeyword(s)
    if not s then return nil end

    for _, k in ipairs(CONFIG.keywords) do
        if string.find(s, k, 1, true) then
            return k
        end
    end

    return nil
end

local function hasFloat1211At(addr)
    local f1 = safeReadFloat(addr)
    local f2 = safeReadFloat(addr + 4)
    local f3 = safeReadFloat(addr + 8)
    local f4 = safeReadFloat(addr + 12)

    return near(f1, 1.0, 0.001)
       and near(f2, 2.0, 0.001)
       and near(f3, 1.0, 0.001)
       and near(f4, 1.0, 0.001)
end

local function scoreHeader(node, first)
    if not isReadable(node) then return 0 end

    local score = 0

    local q00 = safeReadQword(node + 0x00)
    local q08 = safeReadQword(node + 0x08)
    local q10 = safeReadQword(node + 0x10)
    local i18 = safeReadInteger(node + 0x18)
    local q28 = safeReadQword(node + 0x28)
    local q30 = safeReadQword(node + 0x30)

    if isLikelyPtr(q00) then score = score + 18 end
    if isLikelyPtr(q08) then score = score + 8 end
    if isLikelyPtr(q10) then score = score + 18 end
    if i18 and i18 >= 0 and i18 <= 10 then score = score + 16 end
    if isLikelyPtr(q28) then score = score + 8 end
    if isLikelyPtr(q30) then score = score + 8 end

    if first then
        if first.q00 and q00 == first.q00 then score = score + 15 end
        if first.q08 and q08 == first.q08 then score = score + 8 end
        if first.q10 and q10 == first.q10 then score = score + 15 end
    end

    return score
end

local function scoreUsefulBase(base, votes)
    if not isReadable(base) then return 0, "unreadable", {} end

    local score = votes * 80
    local notes = { "votes=" .. tostring(votes) }
    local found = {}
    local details = {}

    local first = {
        q00 = safeReadQword(base + 0x00),
        q08 = safeReadQword(base + 0x08),
        q10 = safeReadQword(base + 0x10),
    }

    local goodHeaders = 0

    for i = 0, CONFIG.checkNodes - 1 do
        local node = base + i * CONFIG.stride
        local hs = scoreHeader(node, first)

        if hs >= 55 then goodHeaders = goodHeaders + 1 end
        score = score + hs
    end

    if goodHeaders >= 4 then
        score = score + 120
        table.insert(notes, "strideOK")
    end

    if goodHeaders >= 8 then
        score = score + 220
        table.insert(notes, "strongStride")
    end

    table.insert(notes, "goodHeaders=" .. tostring(goodHeaders))

    -- Ищем знакомые key/value в первых 42 node.
    for i = 0, 42 do
        local node = base + i * CONFIG.stride
        local _, keyText = readFieldText(node + 0x80)
        local _, valText = readFieldText(node + 0xF8)

        if keyText or valText then
            table.insert(details, {
                index = i,
                node = node,
                key = keyText,
                val = valText,
            })
        end

        if keyText then
            if string.find(keyText, "DefaultTakeButtonText", 1, true) then found.takeText = true end
            if string.find(keyText, "DefaultWasTakenText", 1, true) then found.wasText = true end
            if string.find(keyText, "LevelText", 1, true) then found.levelText = true end
        end

        if valText then
            if string.find(valText, "Забрать", 1, true) then found.takeRu = true end
            if string.find(valText, "Получено", 1, true) then found.received = true end
            if string.find(valText, "Take", 1, true) then found.takeEn = true end
        end
    end

    if found.takeText then score = score + 250; table.insert(notes, "has DefaultTakeButtonText") end
    if found.wasText then score = score + 250; table.insert(notes, "has DefaultWasTakenText") end
    if found.levelText then score = score + 180; table.insert(notes, "has LevelText") end
    if found.takeRu then score = score + 160; table.insert(notes, "has Забрать") end
    if found.received then score = score + 160; table.insert(notes, "has Получено") end
    if found.takeEn then score = score + 80; table.insert(notes, "has Take") end

    return score, table.concat(notes, " | "), details
end

local function locateBase()
    if CONFIG.MANUAL_BASE then
        return CONFIG.MANUAL_BASE, "manual"
    end

    log("============================================================")
    log("STEP 1: LOCATE USEFUL BASE")
    log("============================================================")

    local sl, mode = aobScanFast(CONFIG.floatAob)
    log("scan mode: " .. mode)

    if not sl then
        log("float hits=0")
        return nil, "no_float_hits"
    end

    local hits = {}

    for i = 0, sl.Count - 1 do
        local h = addrFromString(sl[i])
        if h then table.insert(hits, h) end
    end

    sl.destroy()
    table.sort(hits)

    log("float hits=" .. tostring(#hits))

    local voteMap = {}

    local function addVote(base)
        if not base or base < 0x10000 then return end
        base = alignDown(base, 0x8)

        local key = hex(base)

        if not voteMap[key] then
            voteMap[key] = { base = base, votes = 0 }
        end

        voteMap[key].votes = voteMap[key].votes + 1
    end

    for _, hit in ipairs(hits) do
        for _, off in ipairs(CONFIG.possibleFloatOffsets) do
            local nodeGuess = hit - off

            for back = 0, CONFIG.maxStrideBack do
                addVote(nodeGuess - back * CONFIG.stride)
            end
        end
    end

    local candidates = {}

    for _, c in pairs(voteMap) do
        if c.votes >= CONFIG.minVoteHits and isReadable(c.base) then
            table.insert(candidates, c)
        end
    end

    table.sort(candidates, function(a, b)
        if a.votes == b.votes then return a.base < b.base end
        return a.votes > b.votes
    end)

    log("candidates=" .. tostring(#candidates))

    local best = nil

    for i = 1, math.min(#candidates, CONFIG.maxCandidatesToScore) do
        local c = candidates[i]
        local score, notes, details = scoreUsefulBase(c.base, c.votes)

        c.score = score
        c.notes = notes
        c.details = details

        if not best or c.score > best.score then
            best = c
        end
    end

    if not best then
        return nil, "no_candidate"
    end

    log("BEST base=" .. hex(best.base) .. " score=" .. tostring(best.score) .. " notes=" .. tostring(best.notes))

    for _, d in ipairs(best.details or {}) do
        log(string.format(
            "  node[%02d] %s key=%s val=%s",
            d.index,
            hex(d.node),
            tostring(d.key),
            tostring(d.val)
        ))
    end

    addRecord("[useful-base]", best.base)

    return best.base, "auto"
end

------------------------------------------------------------
-- POINTER ANALYSIS
------------------------------------------------------------

local function classifyPtr(ptr, mainStart, mainEnd)
    if not isLikelyPtr(ptr) then return "not_ptr" end
    if ptr >= mainStart and ptr <= mainEnd then return "IN_MAIN_DUMP" end
    if isModulePtr(ptr) then return "MODULE_OR_VTABLE" end
    if isReadable(ptr) then return "READABLE_EXTERNAL" end
    return "UNREADABLE_EXTERNAL"
end

local function objectHeaderScore(addr)
    if not isReadable(addr) then return 0 end

    local score = 0

    local q00 = safeReadQword(addr + 0x00)
    local q08 = safeReadQword(addr + 0x08)
    local q10 = safeReadQword(addr + 0x10)
    local i18 = safeReadInteger(addr + 0x18)
    local q28 = safeReadQword(addr + 0x28)
    local q30 = safeReadQword(addr + 0x30)

    if isLikelyPtr(q00) then score = score + 10 end
    if isModulePtr(q00) then score = score + 15 end
    if isLikelyPtr(q08) then score = score + 6 end
    if isLikelyPtr(q10) then score = score + 8 end
    if i18 and i18 >= 0 and i18 <= 100 then score = score + 6 end
    if isLikelyPtr(q28) then score = score + 6 end
    if isLikelyPtr(q30) then score = score + 6 end

    return score
end

local function pointerDensityScore(addr)
    if not isReadable(addr) then return 0, 0 end

    local count = 0
    local readableCount = 0

    for off = -CONFIG.microBack, CONFIG.microFwd, 8 do
        local q = safeReadQword(addr + off)

        if isLikelyPtr(q) then
            count = count + 1
            if isReadable(q) then
                readableCount = readableCount + 1
            end
        end
    end

    return count, readableCount
end

local function pointsBackIntoMain(addr, mainStart, mainEnd)
    if not isReadable(addr) then return false end

    for off = -CONFIG.microBack, CONFIG.microFwd, 8 do
        local q = safeReadQword(addr + off)

        if q and q >= mainStart and q <= mainEnd then
            return true
        end
    end

    return false
end

local function hasFloatPatternNear(addr)
    if not isReadable(addr) then return false end

    for off = -CONFIG.microBack, CONFIG.microFwd - 0x10, 4 do
        if hasFloat1211At(addr + off) then
            return true
        end
    end

    return false
end

local function analyzePointer(fieldAddr, ptr, mainStart, mainEnd)
    local class = classifyPtr(ptr, mainStart, mainEnd)
    local score = 0
    local notes = {}

    if class == "READABLE_EXTERNAL" then
        score = score + 30
        table.insert(notes, "readable_external")
    elseif class == "IN_MAIN_DUMP" then
        score = score + 10
        table.insert(notes, "inside_main")
    elseif class == "MODULE_OR_VTABLE" then
        score = score + 5
        table.insert(notes, "module/vtable")
    else
        return nil
    end

    local txt = readPtrText(ptr)

    if txt then
        score = score + 80
        table.insert(notes, "text=" .. txt)

        local kw = containsKeyword(txt)

        if kw then
            score = score + 120
            table.insert(notes, "keyword=" .. kw)
        end
    end

    local hs = objectHeaderScore(ptr)
    if hs > 0 then
        score = score + hs
        table.insert(notes, "headerScore=" .. tostring(hs))
    end

    local ptrCount, readablePtrCount = pointerDensityScore(ptr)

    if ptrCount >= 3 then
        score = score + ptrCount * 4
        table.insert(notes, "ptrDensity=" .. tostring(ptrCount))
    end

    if readablePtrCount >= 2 then
        score = score + readablePtrCount * 6
        table.insert(notes, "readablePtrDensity=" .. tostring(readablePtrCount))
    end

    if pointsBackIntoMain(ptr, mainStart, mainEnd) then
        score = score + 80
        table.insert(notes, "pointsBackToMain")
    end

    if hasFloatPatternNear(ptr) then
        score = score + 50
        table.insert(notes, "float1211near")
    end

    return {
        field = fieldAddr,
        ptr = ptr,
        class = class,
        score = score,
        text = txt,
        notes = table.concat(notes, " | "),
    }
end

local function collectOutboundPointers(base)
    local mainStart = base - CONFIG.mainBack
    local mainSize  = CONFIG.mainBack + CONFIG.mainFwd
    local mainEnd   = mainStart + mainSize

    log("")
    log("============================================================")
    log("STEP 2: MAIN DUMP")
    log("============================================================")
    log("base=" .. hex(base))
    log("mainStart=" .. hex(mainStart))
    log("mainEnd=" .. hex(mainEnd))

    dumpRange(mainStart, mainSize, "MAIN DUMP base-512..base+512")

    log("")
    log("============================================================")
    log("STEP 3: OUTBOUND POINTERS")
    log("============================================================")

    local map = {}

    for a = mainStart, mainEnd - 8, 8 do
        local q = safeReadQword(a)

        if isLikelyPtr(q) then
            local p = analyzePointer(a, q, mainStart, mainEnd)

            if p then
                local key = hex(q)

                if not map[key] then
                    map[key] = p
                    map[key].fields = {}
                end

                table.insert(map[key].fields, a)

                if p.score > map[key].score then
                    map[key].score = p.score
                    map[key].notes = p.notes
                end
            end
        end
    end

    local list = {}

    for _, p in pairs(map) do
        table.insert(list, p)
    end

    table.sort(list, function(a, b)
        if a.score == b.score then return a.ptr < b.ptr end
        return a.score > b.score
    end)

    log("unique pointer targets=" .. tostring(#list))

    for i, p in ipairs(list) do
        if i > CONFIG.maxMicroDumps then break end

        log("")
        log(string.format(
            "#%03d score=%d ptr=%s class=%s text=%s",
            i,
            p.score,
            hex(p.ptr),
            p.class,
            tostring(p.text)
        ))

        log("  notes: " .. tostring(p.notes))
        log("  fields:")

        for j = 1, math.min(#p.fields, 12) do
            local field = p.fields[j]
            log(string.format(
                "    %s base%+d",
                hex(field),
                field - base
            ))
        end

        addRecord("[outbound] #" .. tostring(i), p.ptr)
        dumpRange(p.ptr - CONFIG.microBack, CONFIG.microBack + CONFIG.microFwd, "MICRO DUMP #" .. tostring(i) .. " target=" .. hex(p.ptr))
    end

    return list
end

------------------------------------------------------------
-- MAIN
------------------------------------------------------------

log("Useful base + outbound pointer micro-dumper / READ ONLY")
log("Date: " .. os.date("%Y-%m-%d %H:%M:%S"))
log("Output: " .. OUT)
log("")

local base, mode = locateBase()

if not base then
    log("")
    log("FAILED: base not found.")
    log("Reason: " .. tostring(mode))
    log("")
    log("If you already have a working base, set CONFIG.MANUAL_BASE = 0x...")
    log("DONE")
    F:close()
    messageDialog("Base not found.\nSet CONFIG.MANUAL_BASE manually if needed.\n\nReport:\n" .. OUT, mtInformation, mbOK)
    return
end

log("")
log("Using base=" .. hex(base) .. " mode=" .. tostring(mode))

collectOutboundPointers(base)

log("")
log("DONE")
log("Report: " .. OUT)

F:close()

messageDialog("Base dump + pointer micro-dumps complete.\n\nBase: " .. hex(base) .. "\n\nReport:\n" .. OUT, mtInformation, mbOK)
